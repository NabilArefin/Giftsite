     6→     6→type ParticleShape = "heart" | "dot" | "star" | "circle" | "square";
     7→     7→
     8→     8→type Particle = {
     9→     9→  x: number;
    10→    10→  y: number;
    11→    11→  vx: number;
    12→    12→  vy: number;
    13→    13→  size: number;
    14→    14→  life: number;
    15→    15→  age: number;
    16→    16→  color: string;
    17→    17→  shape: ParticleShape;
    18→    18→  spin: number;
    19→    19→  gravity: number;
    20→    20→  wobble: number;
    21→    21→  wobbleSpeed: number;
    22→    22→};
    23→    23→
    24→    24→type CreateOptions = {
    25→    25→  burst?: boolean;
    26→    26→  trail?: boolean;
    27→    27→  confetti?: boolean;
    28→    28→};
    29→    29→
    30→    30→const PALETTE = ["#e34f6f", "#ff8c72", "#f5c85b", "#9ed7c1", "#7b61ff", "#ffffff"];
    31→    31→const AMBIENT_PALETTE = PALETTE.slice(0, PALETTE.length - 1); // exclude white for ambient
    32→    32→
    33→    33→function createParticle(x: number, y: number, opts: CreateOptions = {}): Particle {
    34→    34→  const isBurst = opts.burst || false;
    35→    35→  const isTrail = opts.trail || false;
    36→    36→  const isConfetti = opts.confetti || false;
    37→    37→
    38→    38→  const angle = Math.random() * Math.PI * 2;
    39→    39→  let speed: number;
    40→    40→  if (isBurst) {
    41→    41→    speed = 2.4 + Math.random() * 4.6;
    42→    42→  } else if (isTrail) {
    43→    43→    speed = 0.3 + Math.random() * 0.8;
    44→    44→  } else {
    45→    45→    speed = 0.4 + Math.random() * 1.4;
    46→    46→  }
    47→    47→
    48→    48→  const shapes: ParticleShape[] = isConfetti
    49→    49→    ? ["star", "circle", "square"]
    50→    50→    : ["heart", "dot"];
    51→    51→
    52→    52→  return {
    53→    53→    x,
    54→    54→    y,
    55→    55→    vx: isConfetti
    56→    56→      ? Math.cos(angle) * (1 + Math.random() * 3)
    57→    57→      : Math.cos(angle) * speed,
    58→    58→    vy: isConfetti
    59→    59→      ? Math.sin(angle) * (1 + Math.random() * 3) - 2
    60→    60→      : Math.sin(angle) * speed - (isBurst ? 2 : 0),
    61→    61→    size: isBurst
    62→    62→      ? 5 + Math.random() * 7
    63→    63→      : isTrail
    64→    64→        ? 1.5 + Math.random() * 3
    65→    65→        : isConfetti
    66→    66→          ? 3 + Math.random() * 5
    67→    67→          : 2 + Math.random() * 4,
    68→    68→    life: isBurst
    69→    69→      ? 90 + Math.random() * 50
    70→    70→      : isTrail
    71→    71→        ? 40 + Math.random() * 40
    72→    72→        : isConfetti
    73→    73→          ? 100 + Math.random() * 80
    74→    74→          : 140 + Math.random() * 120,
    75→    75→    age: 0,
    76→    76→    color: isConfetti
    77→    77→      ? PALETTE[Math.floor(Math.random() * PALETTE.length)]
    78→    78→      : AMBIENT_PALETTE[Math.floor(Math.random() * AMBIENT_PALETTE.length)],
    79→    79→    shape: shapes[Math.floor(Math.random() * shapes.length)],
    80→    80→    spin: Math.random() * Math.PI,
    81→    81→    gravity: isConfetti ? 0.06 : isTrail ? 0.01 : 0.018,
    82→    82→    wobble: isConfetti ? 0.08 : 0,
    83→    83→    wobbleSpeed: 0.05 + Math.random() * 0.05,
    84→    84→  };
    85→    85→}

    87→    87→function drawHeart(
    88→    88→  ctx: CanvasRenderingContext2D,
    89→    89→  x: number,
    90→    90→  y: number,
    91→    91→  size: number,
    92→    92→  color: string,
    93→    93→  alpha: number,
    94→    94→  spin: number,
    95→    95→) {
    96→    96→  ctx.save();
    97→    97→  ctx.translate(x, y);
    98→    98→  ctx.rotate(spin);
    99→    99→  ctx.scale(size / 18, size / 18);
   100→   100→  ctx.globalAlpha = alpha;
   101→   101→  ctx.fillStyle = color;
   102→   102→  ctx.beginPath();
   103→   103→  ctx.moveTo(0, 6);
   104→   104→  ctx.bezierCurveTo(-16, -4, -8, -18, 0, -9);
   105→   105→  ctx.bezierCurveTo(8, -18, 16, -4, 0, 6);
   106→   106→  ctx.fill();
   107→   107→  ctx.restore();
   108→   108→}
   109→   109→
   110→   110→function drawStar(
   111→   111→  ctx: CanvasRenderingContext2D,
   112→   112→  x: number,
   113→   113→  y: number,
   114→   114→  size: number,
   115→   115→  color: string,
   116→   116→  alpha: number,
   117→   117→  spin: number,
   118→   118→) {
   119→   119→  ctx.save();
   120→   120→  ctx.translate(x, y);
   121→   121→  ctx.rotate(spin);
   122→   122→  ctx.globalAlpha = alpha;
   123→   123→  ctx.fillStyle = color;
   124→   124→  ctx.beginPath();
   125→   125→  for (let i = 0; i < 5; i++) {
   126→   126→    const outerAngle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
   127→   127→    const innerAngle = outerAngle + (2 * Math.PI) / 10;
   128→   128→    if (i === 0) {
   129→   129→      ctx.moveTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   130→   130→    } else {
   131→   131→      ctx.lineTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   132→   132→    }
   133→   133→    ctx.lineTo(
   134→   134→      Math.cos(innerAngle) * size * 0.45,
   135→   135→      Math.sin(innerAngle) * size * 0.45,
   136→   136→    );
   137→   137→  }
   138→   138→  ctx.closePath();
   139→   139→  ctx.fill();
   140→   140→  ctx.restore();
   141→   141→}
   142→   142→
   143→   143→function drawSquare(
   144→   144→  ctx: CanvasRenderingContext2D,
   145→   145→  x: number,
   146→   146→  y: number,
   147→   147→  size: number,
   148→   148→  color: string,
   149→   149→  alpha: number,
   150→   150→  spin: number,
   151→   151→) {
   152→   152→  ctx.save();
   153→   153→  ctx.translate(x, y);
   154→   154→  ctx.rotate(spin);
   155→   155→  ctx.globalAlpha = alpha;
   156→   156→  ctx.fillStyle = color;
   157→   157→  ctx.fillRect(-size / 2, -size / 2, size, size);
   158→   158→  ctx.restore();
   159→   159→}
   160→   160→
   161→   161→function drawParticle(ctx: CanvasRenderingContext2D, p: Particle) {
   162→   162→  const alpha = Math.max(0, 1 - p.age / p.life);
   163→   163→  if (alpha <= 0) return;
   164→   164→
   165→   165→  switch (p.shape) {
   166→   166→    case "heart":
   167→   167→      drawHeart(ctx, p.x, p.y, p.size * 2.2, p.color, alpha, p.spin);
   168→   168→      break;
   169→   169→    case "star":
   170→   170→      drawStar(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   171→   171→      break;
   172→   172→    case "square":
   173→   173→      drawSquare(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   174→   174→      break;
   175→   175→    default: {
   176→   176→      ctx.globalAlpha = alpha;
   177→   177→      ctx.fillStyle = p.color;
   178→   178→      ctx.beginPath();
   179→   179→      ctx.arc(p.x, p.y, Math.max(0.5, p.size), 0, Math.PI * 2);
   180→   180→      ctx.fill();
   181→   181→      ctx.globalAlpha = 1;
   182→   182→    }
   183→   183→  }
   184→   184→}

   205→   205→
   206→   206→  // ── Cake cutting interaction ──
   207→   207→  // Track drag across the cake to compute cut progress
   208→   208→  useEffect(() => {
   209→   209→    const cakeArea = cakeAreaRef.current;

   211→   211→
   212→   212→    const SEGMENTS = 20; // divide cake width into segments

   215→   215→    const getProgress = () => {
   216→   216→      // Find the longest contiguous run of cut segments
   217→   217→      // A single line covering just 40%+ of the width = full cut
   218→   218→      let current = 0;
   219→   219→      let best = 0;
   220→   220→      for (let i = 0; i < SEGMENTS; i++) {
   221→   221→        if (cutTrackRef.current[i] > 0) {
   222→   222→          current++;
   223→   223→          if (current > best) best = current;
   224→   224→        } else {
   225→   225→          current = 0;
   226→   226→        }
   227→   227→      }
   228→   228→      return Math.min(1, (best / SEGMENTS) / 0.4);

   234→   234→      const x = clientX - rect.left;
   235→   235→      const y = clientY - rect.top;
   236→   236→
   237→   237→      // Check if pointer is within the cut zone (covers the cake body)
   238→   238→      const cutZoneTop = rect.height * 0.18;
   239→   239→      const cutZoneBottom = rect.height * 0.90;
   240→   240→      if (y < cutZoneTop || y > cutZoneBottom) return;
   241→   241→      if (x < 0 || x > rect.width) return;
   242→   242→
   243→   243→      // Mark segment as cut
   244→   244→      const segIndex = Math.floor((x / rect.width) * SEGMENTS);
   245→   245→      if (segIndex >= 0 && segIndex < SEGMENTS) {
   246→   246→        cutTrackRef.current[segIndex] = 1;
   247→   247→      }
   248→   248→
   249→   249→      const progress = getProgress();
   250→   250→      setCutProgress(progress);
   251→   251→      setIsCutting(true);
   252→   252→
   253→   253→      if (progress >= 0.95 && !isCutComplete) {

   267→   267→              particlesRef.current.push(createParticle(cx, h * 0.35, { confetti: true }));
   268→   268→            }

   270→   270→        }, 800);
   271→   271→      }
   272→   272→    };

   282→   282→      }
   283→   283→    };
   284→   284→

   288→   288→

   335→   335→  useEffect(() => {

   431→   431→

   433→   433→    const { w, h } = sizeRef.current;
   434→   434→    const cx = w / 2;
   435→   435→    const cy = h * 0.35;
   436→   436→    const count = w < 768 ? 20 : 40;

   439→   439→    }

   457→   457→
   458→   458→  // Magical wand cursor — smooth follow with easing, velocity tilt, click sparkles.
   459→   459→  useEffect(() => {
   460→   460→    const cursor = cursorRef.current;
   461→   461→    if (!cursor) return;
   462→   462→
   463→   463→    if (window.matchMedia("(hover: none), (pointer: coarse)").matches) {
   464→   464→      cursor.style.display = "none";
   465→   465→      return;
   466→   466→    }
   467→   467→
   468→   468→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   469→   469→    const target = { x: -200, y: -200 };
   470→   470→    const current = { x: -200, y: -200 };
   471→   471→    const velocity = { x: 0, y: 0 };

   473→   473→    let raf = 0;
