     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";

     5→
     6→type ParticleShape = "heart" | "dot" | "star" | "circle" | "square";
     7→
     8→type Particle = {
     9→  x: number;
    10→  y: number;
    11→  vx: number;
    12→  vy: number;
    13→  size: number;
    14→  life: number;
    15→  age: number;
    16→  color: string;
    17→  shape: ParticleShape;
    18→  spin: number;
    19→  gravity: number;
    20→  wobble: number;
    21→  wobbleSpeed: number;
    22→};
    23→
    24→type CreateOptions = {
    25→  burst?: boolean;
    26→  trail?: boolean;
    27→  confetti?: boolean;
    28→};
    29→
    30→const PALETTE = ["#e34f6f", "#ff8c72", "#f5c85b", "#9ed7c1", "#7b61ff", "#ffffff"];
    31→const AMBIENT_PALETTE = PALETTE.slice(0, PALETTE.length - 1); // exclude white for ambient
    32→
    33→function createParticle(x: number, y: number, opts: CreateOptions = {}): Particle {
    34→  const isBurst = opts.burst || false;
    35→  const isTrail = opts.trail || false;
    36→  const isConfetti = opts.confetti || false;
    37→
    38→  const angle = Math.random() * Math.PI * 2;
    39→  let speed: number;
    40→  if (isBurst) {
    41→    speed = 2.4 + Math.random() * 4.6;
    42→  } else if (isTrail) {
    43→    speed = 0.3 + Math.random() * 0.8;
    44→  } else {
    45→    speed = 0.4 + Math.random() * 1.4;
    46→  }
    47→
    48→  const shapes: ParticleShape[] = isConfetti
    49→    ? ["star", "circle", "square"]
    50→    : ["heart", "dot"];
    51→
    52→  return {
    53→    x,
    54→    y,
    55→    vx: isConfetti
    56→      ? Math.cos(angle) * (1 + Math.random() * 3)
    57→      : Math.cos(angle) * speed,
    58→    vy: isConfetti
    59→      ? Math.sin(angle) * (1 + Math.random() * 3) - 2
    60→      : Math.sin(angle) * speed - (isBurst ? 2 : 0),
    61→    size: isBurst
    62→      ? 5 + Math.random() * 7
    63→      : isTrail
    64→        ? 1.5 + Math.random() * 3
    65→        : isConfetti
    66→          ? 3 + Math.random() * 5
    67→          : 2 + Math.random() * 4,
    68→    life: isBurst
    69→      ? 90 + Math.random() * 50
    70→      : isTrail
    71→        ? 40 + Math.random() * 40
    72→        : isConfetti
    73→          ? 100 + Math.random() * 80
    74→          : 140 + Math.random() * 120,
    75→    age: 0,
    76→    color: isConfetti
    77→      ? PALETTE[Math.floor(Math.random() * PALETTE.length)]
    78→      : AMBIENT_PALETTE[Math.floor(Math.random() * AMBIENT_PALETTE.length)],
    79→    shape: shapes[Math.floor(Math.random() * shapes.length)],
    80→    spin: Math.random() * Math.PI,
    81→    gravity: isConfetti ? 0.06 : isTrail ? 0.01 : 0.018,
    82→    wobble: isConfetti ? 0.08 : 0,
    83→    wobbleSpeed: 0.05 + Math.random() * 0.05,
    84→  };
    85→}
    86→
    87→function drawHeart(
    88→  ctx: CanvasRenderingContext2D,
    89→  x: number,
    90→  y: number,
    91→  size: number,
    92→  color: string,
    93→  alpha: number,
    94→  spin: number,
    95→) {
    96→  ctx.save();
    97→  ctx.translate(x, y);
    98→  ctx.rotate(spin);
    99→  ctx.scale(size / 18, size / 18);
   100→  ctx.globalAlpha = alpha;
   101→  ctx.fillStyle = color;
   102→  ctx.beginPath();
   103→  ctx.moveTo(0, 6);
   104→  ctx.bezierCurveTo(-16, -4, -8, -18, 0, -9);
   105→  ctx.bezierCurveTo(8, -18, 16, -4, 0, 6);
   106→  ctx.fill();
   107→  ctx.restore();
   108→}
   109→
   110→function drawStar(
   111→  ctx: CanvasRenderingContext2D,
   112→  x: number,
   113→  y: number,
   114→  size: number,
   115→  color: string,
   116→  alpha: number,
   117→  spin: number,
   118→) {
   119→  ctx.save();
   120→  ctx.translate(x, y);
   121→  ctx.rotate(spin);
   122→  ctx.globalAlpha = alpha;
   123→  ctx.fillStyle = color;
   124→  ctx.beginPath();
   125→  for (let i = 0; i < 5; i++) {
   126→    const outerAngle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
   127→    const innerAngle = outerAngle + (2 * Math.PI) / 10;
   128→    if (i === 0) {
   129→      ctx.moveTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   130→    } else {
   131→      ctx.lineTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   132→    }
   133→    ctx.lineTo(
   134→      Math.cos(innerAngle) * size * 0.45,
   135→      Math.sin(innerAngle) * size * 0.45,
   136→    );
   137→  }
   138→  ctx.closePath();
   139→  ctx.fill();
   140→  ctx.restore();
   141→}
   142→
   143→function drawSquare(
   144→  ctx: CanvasRenderingContext2D,
   145→  x: number,
   146→  y: number,
   147→  size: number,
   148→  color: string,
   149→  alpha: number,
   150→  spin: number,
   151→) {
   152→  ctx.save();
   153→  ctx.translate(x, y);
   154→  ctx.rotate(spin);
   155→  ctx.globalAlpha = alpha;
   156→  ctx.fillStyle = color;
   157→  ctx.fillRect(-size / 2, -size / 2, size, size);
   158→  ctx.restore();
   159→}
   160→
   161→function drawParticle(ctx: CanvasRenderingContext2D, p: Particle) {
   162→  const alpha = Math.max(0, 1 - p.age / p.life);
   163→  if (alpha <= 0) return;
   164→
   165→  switch (p.shape) {
   166→    case "heart":
   167→      drawHeart(ctx, p.x, p.y, p.size * 2.2, p.color, alpha, p.spin);
   168→      break;
   169→    case "star":
   170→      drawStar(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   171→      break;
   172→    case "square":
   173→      drawSquare(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   174→      break;
   175→    default: {
   176→      ctx.globalAlpha = alpha;
   177→      ctx.fillStyle = p.color;
   178→      ctx.beginPath();
   179→      ctx.arc(p.x, p.y, Math.max(0.5, p.size), 0, Math.PI * 2);
   180→      ctx.fill();
   181→      ctx.globalAlpha = 1;
   182→    }
   183→  }
   184→}

   186→export default function GiftPage() {
   187→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   188→  const particlesRef = useRef<Particle[]>([]);
   189→  const rafRef = useRef<number | null>(null);
   190→  const sizeRef = useRef({ w: 0, h: 0 });
   191→  const mouseRef = useRef({ x: -100, y: -100, last: 0 });
   192→  const reducedRef = useRef(false);
   193→  const cursorRef = useRef<HTMLDivElement | null>(null);
   194→
   195→  const [isOpen, setIsOpen] = useState(false);
   196→  const [showCake, setShowCake] = useState(false); // shows cake after Tap Here
   197→  const [cutProgress, setCutProgress] = useState(0);
   198→  const [isCutting, setIsCutting] = useState(false);
   199→  const [isCutComplete, setIsCutComplete] = useState(false);
   200→
   201→  const cakeAreaRef = useRef<HTMLDivElement | null>(null);
   202→  const cutTrackRef = useRef<number[]>([]); // tracks horizontal coverage
   203→  const isDraggingRef = useRef(false);
   204→
   205→
   206→  // ── Cake cutting interaction ──
   207→  // Track drag across the cake to compute cut progress
   208→  useEffect(() => {
   209→    const cakeArea = cakeAreaRef.current;
   210→    if (!cakeArea) return;
   211→
   212→    const SEGMENTS = 20; // divide cake width into segments (lower = easier to cut)
   213→    cutTrackRef.current = new Array(SEGMENTS).fill(0);
   214→
   215→    const getProgress = () => {
   216→      const covered = cutTrackRef.current.filter((v) => v > 0).length;
   217→      return covered / SEGMENTS;
   218→    };
   219→
   220→    const handleMove = (clientX: number, clientY: number) => {
   221→      if (!isDraggingRef.current || isCutComplete) return;
   222→      const rect = cakeArea.getBoundingClientRect();
   223→      const x = clientX - rect.left;
   224→      const y = clientY - rect.top;
   225→
   226→      // Check if pointer is within the cut zone (wider band for easier cutting)
   227→      const cutZoneTop = rect.height * 0.25;
   228→      const cutZoneBottom = rect.height * 0.75;
   229→      if (y < cutZoneTop || y > cutZoneBottom) return;
   230→      if (x < 0 || x > rect.width) return;
   231→
   232→      // Mark segment as cut
   233→      const segIndex = Math.floor((x / rect.width) * SEGMENTS);
   234→      if (segIndex >= 0 && segIndex < SEGMENTS) {
   235→        cutTrackRef.current[segIndex] = 1;
   236→      }
   237→
   238→      const progress = getProgress();
   239→      setCutProgress(progress);
   240→      setIsCutting(true);
   241→
   242→      if (progress >= 0.95 && !isCutComplete) {
   243→        setCutProgress(1);
   244→        setIsCutComplete(true);
   245→        // Trigger the wish after a short delay for the split animation
   246→        setTimeout(() => {
   247→          setIsOpen(true);
   248→          // Burst hearts and confetti inline
   249→          const { w, h } = sizeRef.current;
   250→          const cx = w / 2;
   251→          for (let i = 0; i < 80; i++) {
   252→            particlesRef.current.push(createParticle(cx, h * 0.38, { burst: true }));
   253→          }
   254→          window.setTimeout(() => {
   255→            for (let i = 0; i < 60; i++) {
   256→              particlesRef.current.push(createParticle(cx, h * 0.35, { confetti: true }));
   257→            }
   258→          }, 200);
   259→        }, 800);
   260→      }
   261→    };
   262→
   263→    const onMouseDown = (e: MouseEvent) => {
   264→      const rect = cakeArea.getBoundingClientRect();
   265→      const y = e.clientY - rect.top;
   266→      const cutZoneTop = rect.height * 0.25;
   267→      const cutZoneBottom = rect.height * 0.75;
   268→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   269→        isDraggingRef.current = true;
   270→        handleMove(e.clientX, e.clientY);
   271→      }
   272→    };
   273→
   274→    const onMouseMove = (e: MouseEvent) => {
   275→      handleMove(e.clientX, e.clientY);
   276→    };
   277→
   278→    const onMouseUp = () => {
   279→      isDraggingRef.current = false;
   280→      setIsCutting(false);
   281→    };
   282→
   283→    const onTouchStart = (e: TouchEvent) => {
   284→      const touch = e.touches[0];
   285→      const rect = cakeArea.getBoundingClientRect();
   286→      const y = touch.clientY - rect.top;
   287→      const cutZoneTop = rect.height * 0.25;
   288→      const cutZoneBottom = rect.height * 0.75;
   289→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   290→        isDraggingRef.current = true;
   291→        handleMove(touch.clientX, touch.clientY);
   292→      }
   293→    };
   294→
   295→    const onTouchMove = (e: TouchEvent) => {
   296→      if (!isDraggingRef.current) return;
   297→      e.preventDefault();
   298→      const touch = e.touches[0];
   299→      handleMove(touch.clientX, touch.clientY);
   300→    };
   301→
   302→    const onTouchEnd = () => {
   303→      isDraggingRef.current = false;
   304→      setIsCutting(false);
   305→    };
   306→
   307→    cakeArea.addEventListener("mousedown", onMouseDown);
   308→    window.addEventListener("mousemove", onMouseMove);
   309→    window.addEventListener("mouseup", onMouseUp);
   310→    cakeArea.addEventListener("touchstart", onTouchStart, { passive: true });
   311→    cakeArea.addEventListener("touchmove", onTouchMove, { passive: false });
   312→    cakeArea.addEventListener("touchend", onTouchEnd);
   313→
   314→    return () => {
   315→      cakeArea.removeEventListener("mousedown", onMouseDown);
   316→      window.removeEventListener("mousemove", onMouseMove);
   317→      window.removeEventListener("mouseup", onMouseUp);
   318→      cakeArea.removeEventListener("touchstart", onTouchStart);
   319→      cakeArea.removeEventListener("touchmove", onTouchMove);
   320→      cakeArea.removeEventListener("touchend", onTouchEnd);
   321→    };
   322→  }, [isCutComplete, showCake]);
   323→
   324→  useEffect(() => {
   325→    const canvas = canvasRef.current;
   326→    if (!canvas) return;
   327→    const ctx = canvas.getContext("2d");
   328→    if (!ctx) return;
   329→
   330→    const reducedMotionMQ = window.matchMedia("(prefers-reduced-motion: reduce)");
   331→    reducedRef.current = reducedMotionMQ.matches;
   332→
   333→    const resize = () => {
   334→      const ratio = window.devicePixelRatio || 1;
   335→      const w = window.innerWidth;
   336→      const h = window.innerHeight;
   337→      canvas.width = Math.floor(w * ratio);
   338→      canvas.height = Math.floor(h * ratio);
   339→      canvas.style.width = `${w}px`;
   340→      canvas.style.height = `${h}px`;
   341→      ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
   342→      sizeRef.current = { w, h };
   343→    };
   344→
   345→    resize();
   346→    window.addEventListener("resize", resize);
   347→
   348→    const animate = () => {
   349→      const { w, h } = sizeRef.current;
   350→      ctx.clearRect(0, 0, w, h);
   351→
   352→      // Ambient particles (rising from bottom)
   353→      if (
   354→        !reducedRef.current &&
   355→        particlesRef.current.length < 100 &&
   356→        Math.random() > 0.62
   357→      ) {
   358→        particlesRef.current.push(createParticle(Math.random() * w, h + 20));
   359→      }
   360→
   361→      particlesRef.current = particlesRef.current.filter((p) => p.age < p.life);
   362→
   363→      for (const p of particlesRef.current) {
   364→        p.age += 1;
   365→        p.x += p.vx + Math.sin(p.age * p.wobbleSpeed) * p.wobble;
   366→        p.y += p.vy;
   367→        p.vy += p.gravity;
   368→        p.spin += 0.02;
   369→        drawParticle(ctx, p);
   370→      }
   371→
   372→      rafRef.current = requestAnimationFrame(animate);
   373→    };
   374→
   375→    animate();
   376→
   377→    const onMotionChange = (e: MediaQueryListEvent) => {
   378→      reducedRef.current = e.matches;
   379→    };
   380→    reducedMotionMQ.addEventListener("change", onMotionChange);
   381→
   382→    const onMouseMove = (e: MouseEvent) => {
   383→      mouseRef.current.x = e.clientX;
   384→      mouseRef.current.y = e.clientY;
   385→      if (reducedRef.current) return;
   386→      const now = performance.now();
   387→      if (now - mouseRef.current.last < 40) return; // throttle ~25fps
   388→      mouseRef.current.last = now;
   389→      particlesRef.current.push(
   390→        createParticle(mouseRef.current.x, mouseRef.current.y, { trail: true }),
   391→      );
   392→    };
   393→
   394→    window.addEventListener("mousemove", onMouseMove);
   395→
   396→    return () => {
   397→      window.removeEventListener("resize", resize);
   398→      window.removeEventListener("mousemove", onMouseMove);
   399→      reducedMotionMQ.removeEventListener("change", onMotionChange);
   400→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   401→    };
   402→  }, []);
   403→
   404→  const burstHearts = useCallback(() => {
   405→    const { w, h } = sizeRef.current;
   406→    const cx = w / 2;
   407→    const cy = h * 0.38;
   408→    for (let i = 0; i < 80; i++) {
   409→      particlesRef.current.push(createParticle(cx, cy, { burst: true }));
   410→    }
   411→  }, []);
   412→
   413→  const burstConfetti = useCallback(() => {
   414→    const { w, h } = sizeRef.current;
   415→    const cx = w / 2;
   416→    const cy = h * 0.35;
   417→    for (let i = 0; i < 60; i++) {
   418→      particlesRef.current.push(createParticle(cx, cy, { confetti: true }));
   419→    }
   420→  }, []);
   421→
   422→  const openWish = useCallback(() => {
   423→    setIsOpen(true);
   424→    burstHearts();
   425→    window.setTimeout(burstConfetti, 200);
   426→  }, [burstHearts, burstConfetti]);
   427→
   428→  const closeWish = useCallback(() => setIsOpen(false), []);
   429→
   430→  useEffect(() => {
   431→    const onKey = (e: KeyboardEvent) => {
   432→      if (e.key === "Escape" && isOpen) closeWish();
   433→    };
   434→    window.addEventListener("keydown", onKey);
   435→    return () => window.removeEventListener("keydown", onKey);
   436→  }, [isOpen, closeWish]);
   437→
   438→  // Magical wand cursor — smooth follow with easing, velocity tilt, click sparkles.
   439→  useEffect(() => {
   440→    const cursor = cursorRef.current;
   441→    if (!cursor) return;
   442→
   443→    if (window.matchMedia("(hover: none), (pointer: coarse)").matches) {
   444→      cursor.style.display = "none";
   445→      return;
   446→    }
   447→
   448→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   449→    const target = { x: -200, y: -200 };
   450→    const current = { x: -200, y: -200 };
   451→    const velocity = { x: 0, y: 0 };
   452→    let hasMoved = false;
   453→    let raf = 0;
   454→
   455→    const onMove = (e: MouseEvent) => {
   456→      target.x = e.clientX;
   457→      target.y = e.clientY;
   458→      if (!hasMoved) {
   459→        hasMoved = true;
   460→        current.x = e.clientX;
   461→        current.y = e.clientY;
   462→        cursor.classList.remove("is-hidden");
   463→      }
   464→      const el = e.target as HTMLElement | null;
   465→      if (el && el.closest("button, a, [role='button']")) {
   466→        cursor.classList.add("is-active");
   467→      } else {
   468→        cursor.classList.remove("is-active");
   469→      }
   470→    };
   471→    const onDown = () => {
   472→      cursor.classList.add("is-pressed");
   473→      // Emit sparkle burst on click from wand tip position
   474→      if (!reduced) {
   475→        const tipX = current.x;
   476→        const tipY = current.y;
   477→        for (let i = 0; i < 6; i++) {
   478→          particlesRef.current.push(createParticle(tipX, tipY, { trail: true }));
   479→        }
   480→      }
   481→    };
   482→    const onUp = () => cursor.classList.remove("is-pressed");
   483→    const onLeave = () => cursor.classList.add("is-hidden");
   484→    const onEnter = () => {
   485→      if (hasMoved) cursor.classList.remove("is-hidden");
   486→    };
   487→
   488→    const tick = () => {
   489→      if (reduced) {
   490→        current.x = target.x;
   491→        current.y = target.y;
   492→      } else {
   493→        const prevX = current.x;
   494→        const prevY = current.y;
   495→        current.x += (target.x - current.x) * 0.28;
   496→        current.y += (target.y - current.y) * 0.28;
   497→        velocity.x = current.x - prevX;
   498→        velocity.y = current.y - prevY;
   499→      }
   500→
   501→      // Velocity-based tilt: wand leans in direction of movement
   502→      const speed = Math.sqrt(velocity.x * velocity.x + velocity.y * velocity.y);
   503→      let tiltAngle = -8; // default rest angle
   504→      if (speed > 1.5) {
   505→        const moveAngle = Math.atan2(velocity.y, velocity.x) * (180 / Math.PI);
   506→        // Map movement direction to wand tilt (subtle lean)
   507→        tiltAngle = -8 + velocity.x * 0.8;
   508→        tiltAngle = Math.max(-30, Math.min(20, tiltAngle));
   509→      }
   510→      cursor.style.transform = `translate3d(${current.x - 7}px, ${
   511→        current.y - 7
   512→      }px, 0)`;
   513→      // Apply velocity tilt via CSS custom property
   514→      cursor.style.setProperty("--wand-tilt", `${tiltAngle}deg`);
   515→      raf = requestAnimationFrame(tick);
   516→    };
   517→
   518→    window.addEventListener("mousemove", onMove, { passive: true });
   519→    window.addEventListener("mousedown", onDown);
   520→    window.addEventListener("mouseup", onUp);
   521→    document.addEventListener("mouseleave", onLeave);
   522→    document.addEventListener("mouseenter", onEnter);
   523→    raf = requestAnimationFrame(tick);
   524→
   525→    return () => {
   526→      window.removeEventListener("mousemove", onMove);
   527→      window.removeEventListener("mousedown", onDown);
   528→      window.removeEventListener("mouseup", onUp);
   529→      document.removeEventListener("mouseleave", onLeave);
   530→      document.removeEventListener("mouseenter", onEnter);
   531→      cancelAnimationFrame(raf);
   532→    };
   533→  }, []);
   534→
   535→  return (
   536→    <div className={`gift-root ${isOpen ? "wish-is-open" : ""}`}>
   537→      <div ref={cursorRef} className="magic-cursor is-hidden" aria-hidden="true">
   538→        <svg className="magic-wand" width="48" height="48" viewBox="0 0 90 90" fill="none">
   539→          <defs>
   540→            <linearGradient id="wandShaft" x1="0%" y1="0%" x2="100%" y2="100%">
   541→              <stop offset="0%" stopColor="#d4a056" />
   542→              <stop offset="15%" stopColor="#b8863a" />
   543→              <stop offset="30%" stopColor="#a0722e" />
   544→              <stop offset="50%" stopColor="#7a5420" />
   545→              <stop offset="70%" stopColor="#5c3d14" />
   546→              <stop offset="85%" stopColor="#3d280a" />
   547→              <stop offset="100%" stopColor="#2a1b06" />
   548→            </linearGradient>
   549→            <linearGradient id="wandShaftHighlight" x1="0%" y1="0%" x2="100%" y2="100%">

   552→              <stop offset="100%" stopColor="#8b6914" />

   554→            <linearGradient id="wandHandle" x1="0%" y1="0%" x2="100%" y2="100%">
   555→              <stop offset="0%" stopColor="#4a3520" />
   556→              <stop offset="30%" stopColor="#3a2818" />
   557→              <stop offset="60%" stopColor="#2a1c10" />
   558→              <stop offset="100%" stopColor="#1a0f08" />
