     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import "./gift.css";
     5→
     6→type ParticleShape = "heart" | "dot" | "star" | "circle" | "square";
     7→
     8→type Particle = {
     9→  x: number;
    10→  y: number;
    11→  vx: number;
    12→  vy: number;
    13→  size: number;
    14→  life: number;
    15→  age: number;
    16→  color: string;
    17→  shape: ParticleShape;
    18→  spin: number;
    19→  gravity: number;
    20→  wobble: number;
    21→  wobbleSpeed: number;
    22→};
    23→
    24→type CreateOptions = {
    25→  burst?: boolean;
    26→  trail?: boolean;
    27→  confetti?: boolean;
    28→};
    29→
    30→const PALETTE = ["#e34f6f", "#ff8c72", "#f5c85b", "#9ed7c1", "#7b61ff", "#ffffff"];
    31→const AMBIENT_PALETTE = PALETTE.slice(0, PALETTE.length - 1); // exclude white for ambient
    32→
    33→function createParticle(x: number, y: number, opts: CreateOptions = {}): Particle {
    34→  const isBurst = opts.burst || false;
    35→  const isTrail = opts.trail || false;
    36→  const isConfetti = opts.confetti || false;
    37→
    38→  const angle = Math.random() * Math.PI * 2;
    39→  let speed: number;
    40→  if (isBurst) {
    41→    speed = 2.4 + Math.random() * 4.6;
    42→  } else if (isTrail) {
    43→    speed = 0.3 + Math.random() * 0.8;
    44→  } else {
    45→    speed = 0.4 + Math.random() * 1.4;
    46→  }
    47→
    48→  const shapes: ParticleShape[] = isConfetti
    49→    ? ["star", "circle", "square"]
    50→    : ["heart", "dot"];
    51→
    52→  return {
    53→    x,
    54→    y,
    55→    vx: isConfetti
    56→      ? Math.cos(angle) * (1 + Math.random() * 3)
    57→      : Math.cos(angle) * speed,
    58→    vy: isConfetti
    59→      ? Math.sin(angle) * (1 + Math.random() * 3) - 2
    60→      : Math.sin(angle) * speed - (isBurst ? 2 : 0),
    61→    size: isBurst
    62→      ? 5 + Math.random() * 7
    63→      : isTrail
    64→        ? 1.5 + Math.random() * 3
    65→        : isConfetti
    66→          ? 3 + Math.random() * 5
    67→          : 2 + Math.random() * 4,
    68→    life: isBurst
    69→      ? 90 + Math.random() * 50
    70→      : isTrail
    71→        ? 40 + Math.random() * 40
    72→        : isConfetti
    73→          ? 100 + Math.random() * 80
    74→          : 140 + Math.random() * 120,
    75→    age: 0,
    76→    color: isConfetti
    77→      ? PALETTE[Math.floor(Math.random() * PALETTE.length)]
    78→      : AMBIENT_PALETTE[Math.floor(Math.random() * AMBIENT_PALETTE.length)],
    79→    shape: shapes[Math.floor(Math.random() * shapes.length)],
    80→    spin: Math.random() * Math.PI,
    81→    gravity: isConfetti ? 0.06 : isTrail ? 0.01 : 0.018,
    82→    wobble: isConfetti ? 0.08 : 0,
    83→    wobbleSpeed: 0.05 + Math.random() * 0.05,
    84→  };
    85→}
    86→
    87→function drawHeart(
    88→  ctx: CanvasRenderingContext2D,
    89→  x: number,
    90→  y: number,
    91→  size: number,
    92→  color: string,
    93→  alpha: number,
    94→  spin: number,
    95→) {
    96→  ctx.save();
    97→  ctx.translate(x, y);
    98→  ctx.rotate(spin);
    99→  ctx.scale(size / 18, size / 18);
   100→  ctx.globalAlpha = alpha;
   101→  ctx.fillStyle = color;
   102→  ctx.beginPath();
   103→  ctx.moveTo(0, 6);
   104→  ctx.bezierCurveTo(-16, -4, -8, -18, 0, -9);
   105→  ctx.bezierCurveTo(8, -18, 16, -4, 0, 6);
   106→  ctx.fill();
   107→  ctx.restore();
   108→}
   109→
   110→function drawStar(
   111→  ctx: CanvasRenderingContext2D,
   112→  x: number,
   113→  y: number,
   114→  size: number,
   115→  color: string,
   116→  alpha: number,
   117→  spin: number,
   118→) {
   119→  ctx.save();
   120→  ctx.translate(x, y);
   121→  ctx.rotate(spin);
   122→  ctx.globalAlpha = alpha;
   123→  ctx.fillStyle = color;
   124→  ctx.beginPath();
   125→  for (let i = 0; i < 5; i++) {
   126→    const outerAngle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
   127→    const innerAngle = outerAngle + (2 * Math.PI) / 10;
   128→    if (i === 0) {
   129→      ctx.moveTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   130→    } else {
   131→      ctx.lineTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   132→    }
   133→    ctx.lineTo(
   134→      Math.cos(innerAngle) * size * 0.45,
   135→      Math.sin(innerAngle) * size * 0.45,
   136→    );
   137→  }
   138→  ctx.closePath();
   139→  ctx.fill();
   140→  ctx.restore();
   141→}
   142→
   143→function drawSquare(
   144→  ctx: CanvasRenderingContext2D,
   145→  x: number,
   146→  y: number,
   147→  size: number,
   148→  color: string,
   149→  alpha: number,
   150→  spin: number,
   151→) {
   152→  ctx.save();
   153→  ctx.translate(x, y);
   154→  ctx.rotate(spin);
   155→  ctx.globalAlpha = alpha;
   156→  ctx.fillStyle = color;
   157→  ctx.fillRect(-size / 2, -size / 2, size, size);
   158→  ctx.restore();
   159→}
   160→
   161→function drawParticle(ctx: CanvasRenderingContext2D, p: Particle) {
   162→  const alpha = Math.max(0, 1 - p.age / p.life);
   163→  if (alpha <= 0) return;
   164→
   165→  switch (p.shape) {
   166→    case "heart":
   167→      drawHeart(ctx, p.x, p.y, p.size * 2.2, p.color, alpha, p.spin);
   168→      break;
   169→    case "star":
   170→      drawStar(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   171→      break;
   172→    case "square":
   173→      drawSquare(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   174→      break;
   175→    default: {
   176→      ctx.globalAlpha = alpha;
   177→      ctx.fillStyle = p.color;
   178→      ctx.beginPath();
   179→      ctx.arc(p.x, p.y, Math.max(0.5, p.size), 0, Math.PI * 2);
   180→      ctx.fill();
   181→      ctx.globalAlpha = 1;
   182→    }
   183→  }
   184→}

   186→export default function GiftPage() {
   187→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   188→  const particlesRef = useRef<Particle[]>([]);
   189→  const rafRef = useRef<number | null>(null);
   190→  const sizeRef = useRef({ w: 0, h: 0 });
   191→  const mouseRef = useRef({ x: -100, y: -100, last: 0 });
   192→  const reducedRef = useRef(false);
   193→  const cursorRef = useRef<HTMLDivElement | null>(null);
   194→
   195→  const [isOpen, setIsOpen] = useState(false);
   196→  const [showCake, setShowCake] = useState(false); // shows cake after Tap Here
   197→  const [cutProgress, setCutProgress] = useState(0);
   198→  const [isCutting, setIsCutting] = useState(false);
   199→  const [isCutComplete, setIsCutComplete] = useState(false);
   200→
   201→  const cakeAreaRef = useRef<HTMLDivElement | null>(null);
   202→  const cutTrackRef = useRef<number[]>([]); // tracks horizontal coverage
   203→  const isDraggingRef = useRef(false);
   204→
   205→
   206→  // ── Cake cutting interaction ──
   207→  // Track drag across the cake to compute cut progress
   208→  useEffect(() => {
   209→    const cakeArea = cakeAreaRef.current;
   210→    if (!cakeArea) return;
   211→
   212→    const SEGMENTS = 20; // divide cake width into segments
   213→    cutTrackRef.current = new Array(SEGMENTS).fill(0);
   214→
   215→    const getProgress = () => {
   216→      // Find the longest contiguous run of cut segments
   217→      // A single line covering just 40%+ of the width = full cut
   218→      let current = 0;
   219→      let best = 0;
   220→      for (let i = 0; i < SEGMENTS; i++) {
   221→        if (cutTrackRef.current[i] > 0) {
   222→          current++;
   223→          if (current > best) best = current;
   224→        } else {
   225→          current = 0;
   226→        }
   227→      }
   228→      return Math.min(1, (best / SEGMENTS) / 0.4);
   229→    };
   230→
   231→    const handleMove = (clientX: number, clientY: number) => {
   232→      if (!isDraggingRef.current || isCutComplete) return;
   233→      const rect = cakeArea.getBoundingClientRect();
   234→      const x = clientX - rect.left;
   235→      const y = clientY - rect.top;
   236→
   237→      // Check if pointer is within the cut zone (covers the cake body)
   238→      const cutZoneTop = rect.height * 0.18;
   239→      const cutZoneBottom = rect.height * 0.90;
   240→      if (y < cutZoneTop || y > cutZoneBottom) return;
   241→      if (x < 0 || x > rect.width) return;
   242→
   243→      // Mark segment as cut
   244→      const segIndex = Math.floor((x / rect.width) * SEGMENTS);
   245→      if (segIndex >= 0 && segIndex < SEGMENTS) {
   246→        cutTrackRef.current[segIndex] = 1;
   247→      }
   248→
   249→      const progress = getProgress();
   250→      setCutProgress(progress);
   251→      setIsCutting(true);
   252→
   253→      if (progress >= 0.95 && !isCutComplete) {
   254→        setCutProgress(1);
   255→        setIsCutComplete(true);
   256→        // Trigger the wish after a short delay for the split animation
   257→        setTimeout(() => {
   258→          setIsOpen(true);
   259→          // Burst hearts and confetti inline
   260→          const { w, h } = sizeRef.current;
   261→          const cx = w / 2;
   262→          for (let i = 0; i < 80; i++) {
   263→            particlesRef.current.push(createParticle(cx, h * 0.38, { burst: true }));
   264→          }
   265→          window.setTimeout(() => {
   266→            for (let i = 0; i < 60; i++) {
   267→              particlesRef.current.push(createParticle(cx, h * 0.35, { confetti: true }));
   268→            }
   269→          }, 200);
   270→        }, 800);
   271→      }
   272→    };
   273→
   274→    const onMouseDown = (e: MouseEvent) => {
   275→      const rect = cakeArea.getBoundingClientRect();
   276→      const y = e.clientY - rect.top;
   277→      const cutZoneTop = rect.height * 0.25;
   278→      const cutZoneBottom = rect.height * 0.75;
   279→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   280→        isDraggingRef.current = true;
   281→        handleMove(e.clientX, e.clientY);
   282→      }
   283→    };
   284→
   285→    const onMouseMove = (e: MouseEvent) => {
   286→      handleMove(e.clientX, e.clientY);
   287→    };
   288→
   289→    const onMouseUp = () => {
   290→      isDraggingRef.current = false;
   291→      setIsCutting(false);
   292→    };
   293→
   294→    const onTouchStart = (e: TouchEvent) => {
   295→      const touch = e.touches[0];
   296→      const rect = cakeArea.getBoundingClientRect();
   297→      const y = touch.clientY - rect.top;
   298→      const cutZoneTop = rect.height * 0.25;
   299→      const cutZoneBottom = rect.height * 0.75;
   300→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   301→        isDraggingRef.current = true;
   302→        handleMove(touch.clientX, touch.clientY);
   303→      }
   304→    };
   305→
   306→    const onTouchMove = (e: TouchEvent) => {
   307→      if (!isDraggingRef.current) return;
   308→      e.preventDefault();
   309→      const touch = e.touches[0];
   310→      handleMove(touch.clientX, touch.clientY);
   311→    };
   312→
   313→    const onTouchEnd = () => {
   314→      isDraggingRef.current = false;
   315→      setIsCutting(false);
   316→    };
   317→
   318→    cakeArea.addEventListener("mousedown", onMouseDown);
   319→    window.addEventListener("mousemove", onMouseMove);
   320→    window.addEventListener("mouseup", onMouseUp);
   321→    cakeArea.addEventListener("touchstart", onTouchStart, { passive: true });
   322→    cakeArea.addEventListener("touchmove", onTouchMove, { passive: false });
   323→    cakeArea.addEventListener("touchend", onTouchEnd);
   324→
   325→    return () => {
   326→      cakeArea.removeEventListener("mousedown", onMouseDown);
   327→      window.removeEventListener("mousemove", onMouseMove);
   328→      window.removeEventListener("mouseup", onMouseUp);
   329→      cakeArea.removeEventListener("touchstart", onTouchStart);
   330→      cakeArea.removeEventListener("touchmove", onTouchMove);
   331→      cakeArea.removeEventListener("touchend", onTouchEnd);
   332→    };
   333→  }, [isCutComplete, showCake]);
   334→
   335→  useEffect(() => {
   336→    const canvas = canvasRef.current;
   337→    if (!canvas) return;
   338→    const ctx = canvas.getContext("2d");
   339→    if (!ctx) return;
   340→
   341→    const reducedMotionMQ = window.matchMedia("(prefers-reduced-motion: reduce)");
   342→    reducedRef.current = reducedMotionMQ.matches;
   343→
   344→    const resize = () => {
   345→      const rawRatio = window.devicePixelRatio || 1;
   346→      // Cap pixel ratio at 2 to save GPU on high-DPI mobile
   347→      const ratio = Math.min(rawRatio, 2);
   348→      const w = window.innerWidth;
   349→      const h = window.innerHeight;
   350→      canvas.width = Math.floor(w * ratio);
   351→      canvas.height = Math.floor(h * ratio);
   352→      canvas.style.width = `${w}px`;
   353→      canvas.style.height = `${h}px`;
   354→      ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
   355→      sizeRef.current = { w, h };
   356→    };
   357→
   358→    resize();
   359→    window.addEventListener("resize", resize);
   360→
   361→    const animate = () => {
   362→      const { w, h } = sizeRef.current;
   363→      ctx.clearRect(0, 0, w, h);
   364→
   365→      // Ambient particles (rising from bottom) — fewer on mobile
   366→      const isMobile = w < 768;
   367→      const maxParticles = isMobile ? 30 : 80;
   368→      const spawnChance = isMobile ? 0.88 : 0.62;
   369→      if (
   370→        !reducedRef.current &&
   371→        particlesRef.current.length < maxParticles &&
   372→        Math.random() > spawnChance
   373→      ) {
   374→        particlesRef.current.push(createParticle(Math.random() * w, h + 20));
   375→      }
   376→
   377→      particlesRef.current = particlesRef.current.filter((p) => p.age < p.life);
   378→
   379→      for (const p of particlesRef.current) {
   380→        p.age += 1;
   381→        p.x += p.vx + Math.sin(p.age * p.wobbleSpeed) * p.wobble;
   382→        p.y += p.vy;
   383→        p.vy += p.gravity;
   384→        p.spin += 0.02;
   385→        drawParticle(ctx, p);
   386→      }
   387→
   388→      rafRef.current = requestAnimationFrame(animate);
   389→    };
   390→
   391→    animate();
   392→
   393→    const onMotionChange = (e: MediaQueryListEvent) => {
   394→      reducedRef.current = e.matches;
   395→    };
   396→    reducedMotionMQ.addEventListener("change", onMotionChange);
   397→
   398→    const onMouseMove = (e: MouseEvent) => {
   399→      mouseRef.current.x = e.clientX;
   400→      mouseRef.current.y = e.clientY;
   401→      if (reducedRef.current) return;
   402→      const now = performance.now();
   403→      if (now - mouseRef.current.last < 60) return; // throttle ~16fps for trail
   404→      mouseRef.current.last = now;
   405→      if (particlesRef.current.length < 120) {
   406→        particlesRef.current.push(
   407→          createParticle(mouseRef.current.x, mouseRef.current.y, { trail: true }),
   408→        );
   409→      }
   410→    };
   411→
   412→    window.addEventListener("mousemove", onMouseMove);
   413→
   414→    return () => {
   415→      window.removeEventListener("resize", resize);
   416→      window.removeEventListener("mousemove", onMouseMove);
   417→      reducedMotionMQ.removeEventListener("change", onMotionChange);
   418→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   419→    };
   420→  }, []);
   421→
   422→  const burstHearts = useCallback(() => {
   423→    const { w, h } = sizeRef.current;
   424→    const cx = w / 2;
   425→    const cy = h * 0.38;
   426→    const count = w < 768 ? 30 : 60;
   427→    for (let i = 0; i < count; i++) {
   428→      particlesRef.current.push(createParticle(cx, cy, { burst: true }));
   429→    }
   430→  }, []);
   431→
   432→  const burstConfetti = useCallback(() => {
   433→    const { w, h } = sizeRef.current;
   434→    const cx = w / 2;
   435→    const cy = h * 0.35;
   436→    const count = w < 768 ? 20 : 40;
   437→    for (let i = 0; i < count; i++) {
   438→      particlesRef.current.push(createParticle(cx, cy, { confetti: true }));
   439→    }
   440→  }, []);
   441→
   442→  const openWish = useCallback(() => {
   443→    setIsOpen(true);
   444→    burstHearts();
   445→    window.setTimeout(burstConfetti, 200);
   446→  }, [burstHearts, burstConfetti]);
   447→
   448→  const closeWish = useCallback(() => setIsOpen(false), []);
   449→
   450→  useEffect(() => {
   451→    const onKey = (e: KeyboardEvent) => {
   452→      if (e.key === "Escape" && isOpen) closeWish();
   453→    };
   454→    window.addEventListener("keydown", onKey);
   455→    return () => window.removeEventListener("keydown", onKey);
   456→  }, [isOpen, closeWish]);
   457→
   458→  // Magical wand cursor — smooth follow with easing, velocity tilt, click sparkles.
   459→  useEffect(() => {
   460→    const cursor = cursorRef.current;
   461→    if (!cursor) return;
   462→
   463→    if (window.matchMedia("(hover: none), (pointer: coarse)").matches) {
   464→      cursor.style.display = "none";
   465→      return;
   466→    }
   467→
   468→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   469→    const target = { x: -200, y: -200 };
   470→    const current = { x: -200, y: -200 };
   471→    const velocity = { x: 0, y: 0 };
   472→    let hasMoved = false;
   473→    let raf = 0;
   474→
   475→    const onMove = (e: MouseEvent) => {
   476→      target.x = e.clientX;
   477→      target.y = e.clientY;
   478→      if (!hasMoved) {
   479→        hasMoved = true;
   480→        current.x = e.clientX;
   481→        current.y = e.clientY;
   482→        cursor.classList.remove("is-hidden");
   483→      }
   484→      const el = e.target as HTMLElement | null;
   485→      if (el && el.closest("button, a, [role='button']")) {
   486→        cursor.classList.add("is-active");
   487→      } else {
   488→        cursor.classList.remove("is-active");
   489→      }
   490→    };
   491→    const onDown = () => {
   492→      cursor.classList.add("is-pressed");
   493→      // Emit sparkle burst on click from wand tip position
   494→      if (!reduced) {
   495→        const tipX = current.x;
   496→        const tipY = current.y;
   497→        for (let i = 0; i < 6; i++) {
   498→          particlesRef.current.push(createParticle(tipX, tipY, { trail: true }));
   499→        }
   500→      }
   501→    };
   502→    const onUp = () => cursor.classList.remove("is-pressed");
   503→    const onLeave = () => cursor.classList.add("is-hidden");
   504→    const onEnter = () => {
   505→      if (hasMoved) cursor.classList.remove("is-hidden");
   506→    };
   507→
   508→    const tick = () => {
   509→      if (reduced) {
   510→        current.x = target.x;
   511→        current.y = target.y;
   512→      } else {
   513→        const prevX = current.x;
   514→        const prevY = current.y;
   515→        current.x += (target.x - current.x) * 0.28;
   516→        current.y += (target.y - current.y) * 0.28;
   517→        velocity.x = current.x - prevX;
   518→        velocity.y = current.y - prevY;
   519→      }
   520→
   521→      // Velocity-based tilt: wand leans in direction of movement
   522→      const speed = Math.sqrt(velocity.x * velocity.x + velocity.y * velocity.y);
   523→      let tiltAngle = -8; // default rest angle
   524→      if (speed > 1.5) {
   525→        const moveAngle = Math.atan2(velocity.y, velocity.x) * (180 / Math.PI);
   526→        // Map movement direction to wand tilt (subtle lean)
   527→        tiltAngle = -8 + velocity.x * 0.8;
   528→        tiltAngle = Math.max(-30, Math.min(20, tiltAngle));
   529→      }
   530→      cursor.style.transform = `translate3d(${current.x - 7}px, ${
   531→        current.y - 7
   532→      }px, 0)`;
   533→      // Apply velocity tilt via CSS custom property
   534→      cursor.style.setProperty("--wand-tilt", `${tiltAngle}deg`);
   535→      raf = requestAnimationFrame(tick);
   536→    };
   537→
   538→    window.addEventListener("mousemove", onMove, { passive: true });
   539→    window.addEventListener("mousedown", onDown);
   540→    window.addEventListener("mouseup", onUp);
   541→    document.addEventListener("mouseleave", onLeave);
   542→    document.addEventListener("mouseenter", onEnter);
   543→    raf = requestAnimationFrame(tick);
   544→
   545→    return () => {
   546→      window.removeEventListener("mousemove", onMove);
   547→      window.removeEventListener("mousedown", onDown);
   548→      window.removeEventListener("mouseup", onUp);
   549→      document.removeEventListener("mouseleave", onLeave);
   550→      document.removeEventListener("mouseenter", onEnter);
   551→      cancelAnimationFrame(raf);
   552→    };
   553→  }, []);
   554→
   555→  return (
   556→    <div className={`gift-root ${isOpen ? "wish-is-open" : ""}`}>
   557→      <div ref={cursorRef} className="magic-cursor is-hidden" aria-hidden="true">
   558→        <svg className="magic-wand" width="48" height="48" viewBox="0 0 90 90" fill="none">
   559→          <defs>
   560→            <linearGradient id="wandShaft" x1="0%" y1="0%" x2="100%" y2="100%">
   561→              <stop offset="0%" stopColor="#d4a056" />
   562→              <stop offset="15%" stopColor="#b8863a" />
   563→              <stop offset="30%" stopColor="#a0722e" />
   564→              <stop offset="50%" stopColor="#7a5420" />
   565→              <stop offset="70%" stopColor="#5c3d14" />
