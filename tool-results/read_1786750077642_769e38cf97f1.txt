     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import "./gift.css";
     5→
     6→type ParticleShape = "heart" | "dot" | "star" | "circle" | "square";
     7→
     8→type Particle = {
     9→  x: number;
    10→  y: number;
    11→  vx: number;
    12→  vy: number;
    13→  size: number;
    14→  life: number;
    15→  age: number;
    16→  color: string;
    17→  shape: ParticleShape;
    18→  spin: number;
    19→  gravity: number;
    20→  wobble: number;
    21→  wobbleSpeed: number;
    22→};
    23→
    24→type CreateOptions = {
    25→  burst?: boolean;
    26→  trail?: boolean;
    27→  confetti?: boolean;
    28→};
    29→
    30→const PALETTE = ["#e34f6f", "#ff8c72", "#f5c85b", "#9ed7c1", "#7b61ff", "#ffffff"];
    31→const AMBIENT_PALETTE = PALETTE.slice(0, PALETTE.length - 1); // exclude white for ambient
    32→
    33→function createParticle(x: number, y: number, opts: CreateOptions = {}): Particle {
    34→  const isBurst = opts.burst || false;
    35→  const isTrail = opts.trail || false;
    36→  const isConfetti = opts.confetti || false;
    37→
    38→  const angle = Math.random() * Math.PI * 2;
    39→  let speed: number;
    40→  if (isBurst) {
    41→    speed = 2.4 + Math.random() * 4.6;
    42→  } else if (isTrail) {
    43→    speed = 0.3 + Math.random() * 0.8;
    44→  } else {
    45→    speed = 0.4 + Math.random() * 1.4;
    46→  }
    47→
    48→  const shapes: ParticleShape[] = isConfetti
    49→    ? ["star", "circle", "square"]
    50→    : ["heart", "dot"];
    51→
    52→  return {
    53→    x,
    54→    y,
    55→    vx: isConfetti
    56→      ? Math.cos(angle) * (1 + Math.random() * 3)
    57→      : Math.cos(angle) * speed,
    58→    vy: isConfetti
    59→      ? Math.sin(angle) * (1 + Math.random() * 3) - 2
    60→      : Math.sin(angle) * speed - (isBurst ? 2 : 0),
    61→    size: isBurst
    62→      ? 5 + Math.random() * 7
    63→      : isTrail
    64→        ? 1.5 + Math.random() * 3
    65→        : isConfetti
    66→          ? 3 + Math.random() * 5
    67→          : 2 + Math.random() * 4,
    68→    life: isBurst
    69→      ? 90 + Math.random() * 50
    70→      : isTrail
    71→        ? 40 + Math.random() * 40
    72→        : isConfetti
    73→          ? 100 + Math.random() * 80
    74→          : 140 + Math.random() * 120,
    75→    age: 0,
    76→    color: isConfetti
    77→      ? PALETTE[Math.floor(Math.random() * PALETTE.length)]
    78→      : AMBIENT_PALETTE[Math.floor(Math.random() * AMBIENT_PALETTE.length)],
    79→    shape: shapes[Math.floor(Math.random() * shapes.length)],
    80→    spin: Math.random() * Math.PI,
    81→    gravity: isConfetti ? 0.06 : isTrail ? 0.01 : 0.018,
    82→    wobble: isConfetti ? 0.08 : 0,
    83→    wobbleSpeed: 0.05 + Math.random() * 0.05,
    84→  };
    85→}
    86→
    87→function drawHeart(
    88→  ctx: CanvasRenderingContext2D,
    89→  x: number,
    90→  y: number,
    91→  size: number,
    92→  color: string,
    93→  alpha: number,
    94→  spin: number,
    95→) {
    96→  ctx.save();
    97→  ctx.translate(x, y);
    98→  ctx.rotate(spin);
    99→  ctx.scale(size / 18, size / 18);
   100→  ctx.globalAlpha = alpha;
   101→  ctx.fillStyle = color;
   102→  ctx.beginPath();
   103→  ctx.moveTo(0, 6);
   104→  ctx.bezierCurveTo(-16, -4, -8, -18, 0, -9);
   105→  ctx.bezierCurveTo(8, -18, 16, -4, 0, 6);
   106→  ctx.fill();
   107→  ctx.restore();
   108→}
   109→
   110→function drawStar(
   111→  ctx: CanvasRenderingContext2D,
   112→  x: number,
   113→  y: number,
   114→  size: number,
   115→  color: string,
   116→  alpha: number,
   117→  spin: number,
   118→) {
   119→  ctx.save();
   120→  ctx.translate(x, y);
   121→  ctx.rotate(spin);
   122→  ctx.globalAlpha = alpha;
   123→  ctx.fillStyle = color;
   124→  ctx.beginPath();
   125→  for (let i = 0; i < 5; i++) {
   126→    const outerAngle = (i * 4 * Math.PI) / 5 - Math.PI / 2;
   127→    const innerAngle = outerAngle + (2 * Math.PI) / 10;
   128→    if (i === 0) {
   129→      ctx.moveTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   130→    } else {
   131→      ctx.lineTo(Math.cos(outerAngle) * size, Math.sin(outerAngle) * size);
   132→    }
   133→    ctx.lineTo(
   134→      Math.cos(innerAngle) * size * 0.45,
   135→      Math.sin(innerAngle) * size * 0.45,
   136→    );
   137→  }
   138→  ctx.closePath();
   139→  ctx.fill();
   140→  ctx.restore();
   141→}
   142→
   143→function drawSquare(
   144→  ctx: CanvasRenderingContext2D,
   145→  x: number,
   146→  y: number,
   147→  size: number,
   148→  color: string,
   149→  alpha: number,
   150→  spin: number,
   151→) {
   152→  ctx.save();
   153→  ctx.translate(x, y);
   154→  ctx.rotate(spin);
   155→  ctx.globalAlpha = alpha;
   156→  ctx.fillStyle = color;
   157→  ctx.fillRect(-size / 2, -size / 2, size, size);
   158→  ctx.restore();
   159→}
   160→
   161→function drawParticle(ctx: CanvasRenderingContext2D, p: Particle) {
   162→  const alpha = Math.max(0, 1 - p.age / p.life);
   163→  if (alpha <= 0) return;
   164→
   165→  switch (p.shape) {
   166→    case "heart":
   167→      drawHeart(ctx, p.x, p.y, p.size * 2.2, p.color, alpha, p.spin);
   168→      break;
   169→    case "star":
   170→      drawStar(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   171→      break;
   172→    case "square":
   173→      drawSquare(ctx, p.x, p.y, p.size, p.color, alpha, p.spin);
   174→      break;
   175→    default: {
   176→      ctx.globalAlpha = alpha;
   177→      ctx.fillStyle = p.color;
   178→      ctx.beginPath();
   179→      ctx.arc(p.x, p.y, Math.max(0.5, p.size), 0, Math.PI * 2);
   180→      ctx.fill();
   181→      ctx.globalAlpha = 1;
   182→    }
   183→  }
   184→}

   186→export default function GiftPage() {
   187→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   188→  const particlesRef = useRef<Particle[]>([]);
   189→  const rafRef = useRef<number | null>(null);
   190→  const sizeRef = useRef({ w: 0, h: 0 });
   191→  const mouseRef = useRef({ x: -100, y: -100, last: 0 });
   192→  const reducedRef = useRef(false);
   193→  const cursorRef = useRef<HTMLDivElement | null>(null);
   194→
   195→  const [isOpen, setIsOpen] = useState(false);
   196→  const [showCake, setShowCake] = useState(false); // shows cake after Tap Here
   197→  const [cutProgress, setCutProgress] = useState(0);
   198→  const [isCutting, setIsCutting] = useState(false);
   199→  const [isCutComplete, setIsCutComplete] = useState(false);
   200→
   201→  const cakeAreaRef = useRef<HTMLDivElement | null>(null);
   202→  const cutTrackRef = useRef<number[]>([]); // tracks horizontal coverage
   203→  const isDraggingRef = useRef(false);
   204→
   205→
   206→  // ── Cake cutting interaction ──
   207→  // Track drag across the cake to compute cut progress
   208→  useEffect(() => {
   209→    const cakeArea = cakeAreaRef.current;
   210→    if (!cakeArea) return;
   211→
   212→    const SEGMENTS = 20; // divide cake width into segments
   213→    cutTrackRef.current = new Array(SEGMENTS).fill(0);
   214→
   215→    const getProgress = () => {
   216→      // Find the longest contiguous run of cut segments
   217→      // A single line covering just 40%+ of the width = full cut
   218→      let current = 0;
   219→      let best = 0;
   220→      for (let i = 0; i < SEGMENTS; i++) {
   221→        if (cutTrackRef.current[i] > 0) {
   222→          current++;
   223→          if (current > best) best = current;
   224→        } else {
   225→          current = 0;
   226→        }
   227→      }
   228→      return Math.min(1, (best / SEGMENTS) / 0.4);
   229→    };
   230→
   231→    const handleMove = (clientX: number, clientY: number) => {
   232→      if (!isDraggingRef.current || isCutComplete) return;
   233→      const rect = cakeArea.getBoundingClientRect();
   234→      const x = clientX - rect.left;
   235→      const y = clientY - rect.top;
   236→
   237→      // Check if pointer is within the cut zone (covers the cake body)
   238→      const cutZoneTop = rect.height * 0.18;
   239→      const cutZoneBottom = rect.height * 0.90;
   240→      if (y < cutZoneTop || y > cutZoneBottom) return;
   241→      if (x < 0 || x > rect.width) return;
   242→
   243→      // Mark segment as cut
   244→      const segIndex = Math.floor((x / rect.width) * SEGMENTS);
   245→      if (segIndex >= 0 && segIndex < SEGMENTS) {
   246→        cutTrackRef.current[segIndex] = 1;
   247→      }
   248→
   249→      const progress = getProgress();
   250→      setCutProgress(progress);
   251→      setIsCutting(true);
   252→
   253→      if (progress >= 0.95 && !isCutComplete) {
   254→        setCutProgress(1);
   255→        setIsCutComplete(true);
   256→        // Trigger the wish after a short delay for the split animation
   257→        setTimeout(() => {
   258→          setIsOpen(true);
   259→          // Burst hearts and confetti inline
   260→          const { w, h } = sizeRef.current;
   261→          const cx = w / 2;
   262→          const burstCount = window.innerWidth < 768 ? 25 : 80;
   263→          for (let i = 0; i < burstCount; i++) {
   264→            particlesRef.current.push(createParticle(cx, h * 0.38, { burst: true }));
   265→          }
   266→          const confettiCount = window.innerWidth < 768 ? 15 : 60;
   267→          window.setTimeout(() => {
   268→            for (let i = 0; i < confettiCount; i++) {
   269→              particlesRef.current.push(createParticle(cx, h * 0.35, { confetti: true }));
   270→            }
   271→          }, 200);
   272→        }, 800);
   273→      }
   274→    };
   275→
   276→    const onMouseDown = (e: MouseEvent) => {
   277→      const rect = cakeArea.getBoundingClientRect();
   278→      const y = e.clientY - rect.top;
   279→      const cutZoneTop = rect.height * 0.25;
   280→      const cutZoneBottom = rect.height * 0.75;
   281→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   282→        isDraggingRef.current = true;
   283→        handleMove(e.clientX, e.clientY);
   284→      }
   285→    };
   286→
   287→    const onMouseMove = (e: MouseEvent) => {
   288→      handleMove(e.clientX, e.clientY);
   289→    };
   290→
   291→    const onMouseUp = () => {
   292→      isDraggingRef.current = false;
   293→      setIsCutting(false);
   294→    };
   295→
   296→    const onTouchStart = (e: TouchEvent) => {
   297→      const touch = e.touches[0];
   298→      const rect = cakeArea.getBoundingClientRect();
   299→      const y = touch.clientY - rect.top;
   300→      const cutZoneTop = rect.height * 0.25;
   301→      const cutZoneBottom = rect.height * 0.75;
   302→      if (y >= cutZoneTop && y <= cutZoneBottom) {
   303→        isDraggingRef.current = true;
   304→        handleMove(touch.clientX, touch.clientY);
   305→      }
   306→    };
   307→
   308→    const onTouchMove = (e: TouchEvent) => {
   309→      if (!isDraggingRef.current) return;
   310→      e.preventDefault();
   311→      const touch = e.touches[0];
   312→      handleMove(touch.clientX, touch.clientY);
   313→    };
   314→
   315→    const onTouchEnd = () => {
   316→      isDraggingRef.current = false;
   317→      setIsCutting(false);
   318→    };
   319→
   320→    cakeArea.addEventListener("mousedown", onMouseDown);
   321→    window.addEventListener("mousemove", onMouseMove);
   322→    window.addEventListener("mouseup", onMouseUp);
   323→    cakeArea.addEventListener("touchstart", onTouchStart, { passive: true });
   324→    cakeArea.addEventListener("touchmove", onTouchMove, { passive: false });
   325→    cakeArea.addEventListener("touchend", onTouchEnd);
   326→
   327→    return () => {
   328→      cakeArea.removeEventListener("mousedown", onMouseDown);
   329→      window.removeEventListener("mousemove", onMouseMove);
   330→      window.removeEventListener("mouseup", onMouseUp);
   331→      cakeArea.removeEventListener("touchstart", onTouchStart);
   332→      cakeArea.removeEventListener("touchmove", onTouchMove);
   333→      cakeArea.removeEventListener("touchend", onTouchEnd);
   334→    };
   335→  }, [isCutComplete, showCake]);
   336→
   337→  useEffect(() => {
   338→    const canvas = canvasRef.current;
   339→    if (!canvas) return;
   340→    const ctx = canvas.getContext("2d");
   341→    if (!ctx) return;
   342→
   343→    const reducedMotionMQ = window.matchMedia("(prefers-reduced-motion: reduce)");
   344→    reducedRef.current = reducedMotionMQ.matches;
   345→
   346→    const isMobileDevice = window.innerWidth < 768;
   347→
   348→    const resize = () => {
   349→      const rawRatio = window.devicePixelRatio || 1;
   350→      // Cap pixel ratio to save GPU — 1 on mobile, 2 on desktop
   351→      const ratio = isMobileDevice ? 1 : Math.min(rawRatio, 2);
   352→      const w = window.innerWidth;
   353→      const h = window.innerHeight;
   354→      canvas.width = Math.floor(w * ratio);
   355→      canvas.height = Math.floor(h * ratio);
   356→      canvas.style.width = `${w}px`;
   357→      canvas.style.height = `${h}px`;
   358→      ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
   359→      sizeRef.current = { w, h };
   360→    };
   361→
   362→    resize();
   363→    window.addEventListener("resize", resize);
   364→
   365→    // Throttle frame rate on mobile to reduce GPU load
   366→    let lastFrameTime = 0;
   367→    const frameInterval = isMobileDevice ? 50 : 16; // ~20fps on mobile, ~60fps on desktop
   368→
   369→    const animate = (timestamp: number) => {
   370→      // Frame throttling — skip frames that arrive too soon
   371→      const delta = timestamp - lastFrameTime;
   372→      if (delta < frameInterval) {
   373→        rafRef.current = requestAnimationFrame(animate);
   374→        return;
   375→      }
   376→      lastFrameTime = timestamp - (delta % frameInterval);
   377→
   378→      const { w, h } = sizeRef.current;
   379→      ctx.clearRect(0, 0, w, h);
   380→
   381→      // Ambient particles (rising from bottom) — fewer on mobile
   382→      const isMobile = w < 768;
   383→      const maxParticles = isMobile ? 8 : 80;
   384→      const spawnChance = isMobile ? 0.97 : 0.62;
   385→      if (
   386→        !reducedRef.current &&
   387→        particlesRef.current.length < maxParticles &&
   388→        Math.random() > spawnChance
   389→      ) {
   390→        particlesRef.current.push(createParticle(Math.random() * w, h + 20));
   391→      }
   392→
   393→      particlesRef.current = particlesRef.current.filter((p) => p.age < p.life);
   394→
   395→      for (const p of particlesRef.current) {
   396→        p.age += 1;
   397→        p.x += p.vx + Math.sin(p.age * p.wobbleSpeed) * p.wobble;
   398→        p.y += p.vy;
   399→        p.vy += p.gravity;
   400→        p.spin += 0.02;
   401→        drawParticle(ctx, p);
   402→      }
   403→
   404→      rafRef.current = requestAnimationFrame(animate);
   405→    };
   406→
   407→    animate();
   408→
   409→    const onMotionChange = (e: MediaQueryListEvent) => {
   410→      reducedRef.current = e.matches;
   411→    };
   412→    reducedMotionMQ.addEventListener("change", onMotionChange);
   413→
   414→    const onMouseMove = (e: MouseEvent) => {
   415→      mouseRef.current.x = e.clientX;
   416→      mouseRef.current.y = e.clientY;
   417→      if (reducedRef.current) return;
   418→      const now = performance.now();
   419→      if (now - mouseRef.current.last < (isMobileDevice ? 200 : 80)) return; // throttle trail particles
   420→      mouseRef.current.last = now;
   421→      const maxTrail = isMobileDevice ? 15 : 120;
   422→      if (particlesRef.current.length < maxTrail) {
   423→        particlesRef.current.push(
   424→          createParticle(mouseRef.current.x, mouseRef.current.y, { trail: true }),
   425→        );
   426→      }
   427→    };
   428→
   429→    window.addEventListener("mousemove", onMouseMove);
   430→
   431→    return () => {
   432→      window.removeEventListener("resize", resize);
   433→      window.removeEventListener("mousemove", onMouseMove);
   434→      reducedMotionMQ.removeEventListener("change", onMotionChange);
   435→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   436→    };
   437→  }, []);
   438→
   439→  const burstHearts = useCallback(() => {
   440→    const { w, h } = sizeRef.current;
   441→    const cx = w / 2;
   442→    const cy = h * 0.38;
   443→    const count = w < 768 ? 10 : 60;
   444→    for (let i = 0; i < count; i++) {
   445→      particlesRef.current.push(createParticle(cx, cy, { burst: true }));
   446→    }
   447→  }, []);
   448→
   449→  const burstConfetti = useCallback(() => {
   450→    const { w, h } = sizeRef.current;
   451→    const cx = w / 2;
   452→    const cy = h * 0.35;
   453→    const count = w < 768 ? 6 : 40;
   454→    for (let i = 0; i < count; i++) {
   455→      particlesRef.current.push(createParticle(cx, cy, { confetti: true }));
   456→    }
   457→  }, []);
   458→
   459→  const openWish = useCallback(() => {
   460→    setIsOpen(true);
   461→    burstHearts();
   462→    window.setTimeout(burstConfetti, 200);
   463→  }, [burstHearts, burstConfetti]);
   464→
   465→  const closeWish = useCallback(() => setIsOpen(false), []);
   466→
   467→  useEffect(() => {
   468→    const onKey = (e: KeyboardEvent) => {
   469→      if (e.key === "Escape" && isOpen) closeWish();
   470→    };
   471→    window.addEventListener("keydown", onKey);
   472→    return () => window.removeEventListener("keydown", onKey);
   473→  }, [isOpen, closeWish]);
   474→
   475→  // Magical wand cursor — smooth follow with easing, velocity tilt, click sparkles.
   476→  useEffect(() => {
   477→    const cursor = cursorRef.current;
   478→    if (!cursor) return;
   479→
   480→    if (window.matchMedia("(hover: none), (pointer: coarse)").matches) {
   481→      cursor.style.display = "none";
   482→      return;
   483→    }
   484→
   485→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   486→    const target = { x: -200, y: -200 };
   487→    const current = { x: -200, y: -200 };
   488→    const velocity = { x: 0, y: 0 };
   489→    let hasMoved = false;
   490→    let raf = 0;
   491→
   492→    const onMove = (e: MouseEvent) => {
   493→      target.x = e.clientX;
   494→      target.y = e.clientY;
   495→      if (!hasMoved) {
   496→        hasMoved = true;
   497→        current.x = e.clientX;
   498→        current.y = e.clientY;
   499→        cursor.classList.remove("is-hidden");
   500→      }
   501→      const el = e.target as HTMLElement | null;
   502→      if (el && el.closest("button, a, [role='button']")) {
   503→        cursor.classList.add("is-active");
   504→      } else {
   505→        cursor.classList.remove("is-active");
   506→      }
   507→    };
   508→    const onDown = () => {
   509→      cursor.classList.add("is-pressed");
   510→      // Emit sparkle burst on click from wand tip position
   511→      if (!reduced) {
   512→        const tipX = current.x;
   513→        const tipY = current.y;
   514→        for (let i = 0; i < 6; i++) {
   515→          particlesRef.current.push(createParticle(tipX, tipY, { trail: true }));
   516→        }
   517→      }
   518→    };
   519→    const onUp = () => cursor.classList.remove("is-pressed");
   520→    const onLeave = () => cursor.classList.add("is-hidden");
   521→    const onEnter = () => {
   522→      if (hasMoved) cursor.classList.remove("is-hidden");
   523→    };
   524→
   525→    const tick = () => {
   526→      if (reduced) {
   527→        current.x = target.x;
   528→        current.y = target.y;
   529→      } else {
   530→        const prevX = current.x;
   531→        const prevY = current.y;
   532→        current.x += (target.x - current.x) * 0.55;
   533→        current.y += (target.y - current.y) * 0.55;
   534→        velocity.x = current.x - prevX;
   535→        velocity.y = current.y - prevY;
   536→      }
   537→
   538→      // Velocity-based tilt: wand leans in direction of movement
   539→      const speed = Math.sqrt(velocity.x * velocity.x + velocity.y * velocity.y);
   540→      let tiltAngle = -8; // default rest angle
   541→      if (speed > 1.5) {
   542→        const moveAngle = Math.atan2(velocity.y, velocity.x) * (180 / Math.PI);
   543→        // Map movement direction to wand tilt (subtle lean)
   544→        tiltAngle = -8 + velocity.x * 0.8;
   545→        tiltAngle = Math.max(-30, Math.min(20, tiltAngle));
   546→      }
   547→      cursor.style.transform = `translate3d(${current.x - 7}px, ${
   548→        current.y - 7
   549→      }px, 0)`;
   550→      // Apply velocity tilt via CSS custom property
   551→      cursor.style.setProperty("--wand-tilt", `${tiltAngle}deg`);
   552→      raf = requestAnimationFrame(tick);
   553→    };
   554→
   555→    window.addEventListener("mousemove", onMove, { passive: true });
   556→    window.addEventListener("mousedown", onDown);
   557→    window.addEventListener("mouseup", onUp);
   558→    document.addEventListener("mouseleave", onLeave);
   559→    document.addEventListener("mouseenter", onEnter);
   560→    raf = requestAnimationFrame(tick);
   561→
   562→    return () => {
   563→      window.removeEventListener("mousemove", onMove);
   564→      window.removeEventListener("mousedown", onDown);
   565→      window.removeEventListener("mouseup", onUp);
   566→      document.removeEventListener("mouseleave", onLeave);
   567→      document.removeEventListener("mouseenter", onEnter);
   568→      cancelAnimationFrame(raf);
   569→    };
   570→  }, []);
